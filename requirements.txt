# SLOPANOC backend runtime dependencies.
#
# This is the smallest committed manifest that reproduces the direct
# packages backend/ actually imports/needs to run, audited against the
# real `import`/`from ... import` statements in backend/ (excluding
# backend/tests/) rather than a `pip freeze` of the whole developer
# environment. Versions are pinned to what this environment currently
# runs against (proven by the full backend test suite), so upgrading any
# of these -- especially google-adk/google-genai, which must stay a
# mutually compatible pair -- is a deliberate, separately-tested change,
# not a side effect of running `pip install -r requirements.txt`.
#
# Does NOT include cloud-sql-python-connector: local development uses the
# Cloud SQL Auth Proxy (see README.md), not the Python Connector.
#
# Test-only tooling (pytest, pytest-asyncio) lives in requirements-dev.txt.

# --- Web framework / ASGI ---------------------------------------------
fastapi==0.136.1
starlette==0.52.1
uvicorn==0.46.0
# Required by FastAPI/Starlette for multipart/form-data request parsing --
# the chat attachment upload endpoint (POST-5.1 B2,
# backend/api/app.py's attachment upload route) is this backend's first
# direct use of `UploadFile`/multipart requests. Pinned to a current
# secure release, not the environment's pre-existing older transitive
# version -- this parser sits directly on the untrusted upload boundary.
python-multipart==0.0.32

# --- Data validation ----------------------------------------------------
pydantic==2.13.4

# --- Database / persistence ---------------------------------------------
# SQLAlchemy async engine, used by ADK's own DatabaseSessionService,
# backend/cases/db.py, and backend/knowledge/repository/sqlalchemy.py.
sqlalchemy==2.0.49
# DBAPI driver for the local-development default (sqlite+aiosqlite://...).
aiosqlite==0.22.1
# DBAPI driver for postgresql+asyncpg://... (Cloud SQL PostgreSQL).
asyncpg==0.31.0
# Schema migrations for SLOPANOC-owned tables (Case/Fault, Governed
# Knowledge, Chat Attachments) -- see alembic/. Never manages ADK's own
# session tables.
alembic==1.18.4

# --- Google Cloud / Gemini -----------------------------------------------
google-adk==1.33.0
google-genai==1.75.0
google-cloud-secret-manager==2.28.0
# Private GCS storage for durable chat attachment binaries (POST-5.1 B1) --
# backend/attachments/storage.py. Metadata/reference only reaches Cloud
# SQL; the binary itself lives here.
google-cloud-storage==3.10.1

# --- HTTP client ----------------------------------------------------------
# The Power Automate gateway client (backend/gateway/power_automate_client.py).
requests==2.33.1

# --- Image validation -----------------------------------------------------
# POST-5.1 B2: decode/verify actual uploaded image bytes against the
# claimed MIME type (backend/attachments/validation.py) -- never trust a
# filename extension or a client-supplied Content-Type alone. Pinned to a
# current release (never the environment's pre-existing older transitive
# version) precisely because this library decodes untrusted, unauthenticated-
# boundary input (webp decode vulnerabilities fixed after 10.0.1 are
# directly relevant given image/webp is an accepted type here).
Pillow==12.3.0

# --- Knowledge ingestion / compound-document extraction (A5) --------------
# Mature, focused, single-purpose parsers only -- deliberately not a
# broad document-processing framework. None of these execute macros,
# VBA, spreadsheet formulas, or embedded scripts; all read structured
# data only (backend/knowledge/ingestion/extractors/).
#
# DOCX text/headings/tables/relationships (used to also discover embedded
# media/objects via the OOXML package the library already parses).
python-docx==1.2.0
# PDF page text/page count -- a mature, bounded parser; never used for
# OCR and never executes embedded PDF actions/scripts (it doesn't have
# that capability at all).
pypdf==6.15.0
# XLSX workbook/sheet/cell structure -- formulas are read as their
# literal string form (data_only=False) and NEVER evaluated.
openpyxl==3.1.2
# Legacy OLE2 Compound-File-Binary embedded-object reading (A5
# corrective pass) -- e.g. a Word "Insert Object > Create from File"
# embed, distinct from a native OOXML word/embeddings/*.docx package.
# A narrow, mature, pure-Python container reader with NO code-execution
# capability of any kind (no VBA/macro interpreter, no COM/automation) --
# deliberately not a broader office-automation/forensics framework.
olefile==0.47
